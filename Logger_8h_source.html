27 static std::string
28 GenerateTimestampedFilename(const std::string& baseLogPath)
29 {
30 auto now = std::chrono::system_clock::now();
31 auto time = std::chrono::system_clock::to_time_t(now);
32
33 std::stringstream ss;
34 ss << std::put_time(std::localtime(&time), "%Y-%m-%d_%H-%M-%S");
35
36 // Split the path into directory, name, and extension
37 size_t lastSlash = baseLogPath.find_last_of("/\\");
38 size_t lastDot = baseLogPath.find_last_of('.');
39
40 std::string directory
41 = (lastSlash != std::string::npos)
42 ? baseLogPath.substr(0, lastSlash + 1)
43 : "";
44 std::string filename = (lastSlash != std::string::npos)
45 ? baseLogPath.substr(lastSlash + 1)
46 : baseLogPath;
47 std::string extension
48 = (lastDot != std::string::npos && lastDot > lastSlash)
49 ? baseLogPath.substr(lastDot)
50 : "";
51 std::string basename
52 = (lastDot != std::string::npos && lastDot > lastSlash)
53 ? filename.substr(0, lastDot - (lastSlash + 1))
54 : filename;
55
56 return directory + basename + "_" + ss.str() + extension;
57 }

98 static void
99 Init(const std::string& loggerName = "async_logger",
100 const std::string& logFilePath = "logs/app.log",
101 const bool enableConsole = true, const bool enableFile = true,
102 const spdlog::level::level_enum globalLevel = spdlog::level::trace,
103 const size_t queueSize = 8192, const size_t numThreads = 1,
104 size_t maxFileSize = 1024 * 1024 * 300, size_t maxFiles = 5,
105 spdlog::sink_ptr customSink = nullptr)
106 {
107 std::unique_lock<std::shared_mutex> lock(logMutex());
108 if (spdlog::get(loggerName)) { return; }
109
110 if (!spdlog::thread_pool()) {
111 spdlog::init_thread_pool(queueSize, numThreads);
112 }
113 std::vector<spdlog::sink_ptr> sinks;
114
115 if (enableConsole) {
116 auto consoleSink = std::make_shared<
117 spdlog::sinks::stdout_color_sink_mt>();
118 consoleSink->set_level(globalLevel);
119 sinks.push_back(consoleSink);
120 }
121
122 if (enableFile) {
123 std::string timestampedPath
124 = GenerateTimestampedFilename(logFilePath);
125 auto fileSink = std::make_shared<
126 spdlog::sinks::rotating_file_sink_mt>(
127 timestampedPath, maxFileSize, maxFiles);
128 fileSink->set_level(globalLevel);
129 sinks.push_back(fileSink);
130 }
131
132 if (customSink) { sinks.push_back(customSink); }
133
134 if (sinks.empty()) {
135 throw std::runtime_error(
136 "Logger requires at least one sink (file or console).");
137 }
138
139 auto asyncLogger = std::make_shared<spdlog::async_logger>(
140 loggerName, sinks.begin(), sinks.end(),
141 spdlog::thread_pool(),
142 spdlog::async_overflow_policy::block);
143
144 asyncLogger->set_level(globalLevel);
145 asyncLogger->set_pattern(
146 "[%Y-%m-%d %H:%M:%S.%e] [t %t] [%^%l%$] [%s:%#] %v");
147
148 spdlog::register_logger(asyncLogger);
149 spdlog::set_default_logger(asyncLogger);
150 spdlog::set_level(globalLevel);
151 spdlog::set_error_handler([](const std::string& msg) {
152 fprintf(stderr, "SPDLOG INTERNAL ERROR: %s\n", msg.c_str());
153 });
154 }
